cmake_minimum_required(VERSION 3.22)
project(offline_mapping_optimizer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(MAP_OPTIMIZER_USE_CUDA "Enable CUDA raycasting" OFF)
option(MAP_OPTIMIZER_USE_ABSL "Enable absl hash maps" OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(MPI REQUIRED COMPONENTS C)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree surface)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(GTSAM 4.2 CONFIG REQUIRED)
if(GTSAM_VERSION VERSION_GREATER_EQUAL 4.3)
  message(FATAL_ERROR "yang_nav requires GTSAM 4.2.x; select its GTSAM_DIR")
endif()
find_package(gtsam_points 1.2.0 CONFIG QUIET)
if(NOT gtsam_points_FOUND)
  get_filename_component(LOCAL_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../../.deps/offline_optimizer/install" ABSOLUTE)
  find_package(gtsam_points 1.2.0 CONFIG REQUIRED PATHS "${LOCAL_DEPS}" NO_DEFAULT_PATH)
endif()
add_executable(offline_mapping_optimizer_node
  src/offline_mapping_optimizer_node.cpp src/pose_optimizer.cpp src/raycasting_filter.cpp)
target_include_directories(offline_mapping_optimizer_node PRIVATE include ${PCL_INCLUDE_DIRS})
ament_target_dependencies(offline_mapping_optimizer_node rclcpp)
target_link_libraries(offline_mapping_optimizer_node
  ${PCL_LIBRARIES} Eigen3::Eigen OpenMP::OpenMP_CXX gtsam gtsam_points::gtsam_points)
target_compile_options(offline_mapping_optimizer_node PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(offline_mapping_optimizer_node PROPERTIES
  BUILD_RPATH "$<TARGET_FILE_DIR:gtsam>;$<TARGET_FILE_DIR:gtsam_points::gtsam_points>"
  INSTALL_RPATH "$<TARGET_FILE_DIR:gtsam>;$<TARGET_FILE_DIR:gtsam_points::gtsam_points>")
target_link_options(offline_mapping_optimizer_node PRIVATE "LINKER:--disable-new-dtags")
if(MAP_OPTIMIZER_USE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(offline_mapping_optimizer_node PRIVATE src/raycasting_cuda.cu)
  target_compile_definitions(offline_mapping_optimizer_node PRIVATE MAP_OPTIMIZER_USE_CUDA=1)
  set_target_properties(offline_mapping_optimizer_node PROPERTIES
    CUDA_SEPARABLE_COMPILATION ON CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(offline_mapping_optimizer_node CUDA::cudart)
endif()
if(MAP_OPTIMIZER_USE_ABSL)
  find_package(absl REQUIRED)
  target_link_libraries(offline_mapping_optimizer_node absl::base absl::flat_hash_map)
  target_compile_definitions(offline_mapping_optimizer_node PRIVATE MAP_OPTIMIZER_USE_ABSL=1)
endif()
install(TARGETS offline_mapping_optimizer_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/optimize_map.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_map_bundle test/test_map_bundle.py TIMEOUT 240)
endif()
ament_package()
